git init - инициализация репозитория
git status - узнать статус репозитория
git add file - добавить файл в индекс (. для всех файлов)
git commit -m "text" - фиксация изменений
git branch "text" - создание новой ветки
git branch -r - список веток
git switch "text" - переход на ветку
git remote add origin "url" - связать удаленный репозиторий с локальным
git remote -v - показывает связи с удаленным репозиторием
git push origin "название ветки" - отправление данных в репозиторий
git fetch - загрузить изменения из удаленного репозитория в локальную копию без фиксации в локальном репозитории
git log "ветка"..origin/"ветка" - сравнение изменений
git diff "ветка" origin/"ветка" - различия в файлах
git merge origin/"ветка" - слияние удаленной ветки в локальную
git merge "ветка" - слияние локальных веток
git pull - загрузка изменений из удаленного репозитория и слияние удаленной ветки с локальной
